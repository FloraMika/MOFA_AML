factors <- read.csv("processing/MOFA_transcriptomics_scores.csv", row.names = 1)

load("/home/flomik/Data/MOFA_R_Objects/AML_MAexp_v05.RData")
drug <- data.frame(MAexp@ExperimentList@listData$DSRT@assays@data)
drug <- data.frame(t(drug[,-c(1:2)]))
drug$patient_id <- rownames(drug)

data <- merge(factors, drug, by = "patient_id")
a <- ggplot(data = data, mapping = aes(x = Factor2_score, y = FIMM115484)) +
          geom_point(size = 1) + xlab("DDHM2 score") + ylab("Venetoclax response")+
          sm_statCorr(corr_method = "pearson", size = 0.1) +
