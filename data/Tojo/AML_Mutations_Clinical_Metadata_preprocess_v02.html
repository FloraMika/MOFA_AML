Create variants.df, containting the variants detected for each gene and patient.

ClinSeq_Curated_mutations_AML %>% 
  select(patient_id, feature_ID, Gene_Name) %>% 
  group_by(patient_id, Gene_Name) %>% 
  summarise(variants = paste(feature_ID, collapse = ";")) %>%
  spread(key=Gene_Name, value = variants) %>%
  right_join(data.frame(patient_id=mutations.df$patient_id)) %>% 
  mutate(patient_id = as.character(patient_id)) %>% 
  arrange(patient_id) -> variants.df
## `summarise()` regrouping by 'patient_id' (override with `.groups` argument)
